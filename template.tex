\texttt{cmc-article} includes helpful math packages: \texttt{mathtools},
\texttt{amssymb}, \texttt{amsthm}, and \texttt{physics} by default. It
also includes a default \texttt{header.tex} file with useful macros for

\begin{equation}\protect\phantomsection\label{eq-first}{
\begin{aligned}
   \mathbf{X} &\sim \mathcal{N}(\boldsymbol{\mu}, \boldsymbol{\Sigma}^2); \quad
    p(\mathbf{x}) = \frac{1}{\sqrt{(2\pi)^k\det(\mathbf{*}\Sigma)}}
        \exp(-\frac{1}{2} (\mathbf{x}-\boldsymbol{\mu})^\top \boldsymbol{\Sigma}^{-1} (\mathbf{x}-\boldsymbol{\mu})) \\
    \mathop{\mathrm{\mathbb{E}}}(Y) &= \sum_{y\in\mathcal{Y}} y\mathop{\mathrm{\mathbb{P}}}(Y=y)
    = \sum_{y\in\mathcal{Y}} y\mathop{\mathrm{\mathbb{E}}}(\mathbb{1}\{Y=y\})
\end{aligned}
}\end{equation}

The package also includes an \texttt{assump} environment for typesetting
assumptions which can be referenced by easy-to-remember abbreviations.

\begin{assump}{IID}
The observations $X_1,X_2,\dots,X_n$ are independent and identically distributed.
\end{assump}

\begin{assump}[Finite expectation]{FE}
For each $1\le i\le n$, we have $\mathop{\mathrm{\mathbb{E}}}[X_i]=\mu<\infty$.
\end{assump}

\begin{theorem}[Weak Law of Large
Numbers]\protect\hypertarget{thm-wlln}{}\label{thm-wlln}

Let \(\bar X_n \coloneq n^{-1}\sum_{i=1}^n X_i\). Then under \ref{a-iid}
and \ref{a-fe}, we have \(\bar X_n\xrightarrow{\:p\,}\mu\) as
\(n\to\infty\).

\end{theorem}
